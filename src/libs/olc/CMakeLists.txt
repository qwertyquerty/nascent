message(STATUS "Building olcPixelGameEngine")
add_library(olcPixelGameEngine STATIC olcPixelGameEngine.cpp)

target_include_directories(olcPixelGameEngine SYSTEM PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/../olcPixelGameEngine)
target_link_libraries(olcPixelGameEngine PUBLIC miniaudio freetype Threads::Threads)

if(WIN32)
    target_link_libraries(olcPixelGameEngine PUBLIC user32 gdi32 gdiplus opengl32 dwmapi shlwapi)
elseif(APPLE)
    find_package(OpenGL REQUIRED)
    find_package(GLUT REQUIRED)
    target_compile_definitions(olcPixelGameEngine PUBLIC OLC_IMAGE_STB GL_SILENCE_DEPRECATION)
    target_link_libraries(olcPixelGameEngine PRIVATE stb)
    target_link_libraries(olcPixelGameEngine PUBLIC OpenGL::GL GLUT::GLUT "-framework Carbon" objc)
else()
    find_package(X11 REQUIRED)
    find_package(OpenGL REQUIRED)
    target_compile_definitions(olcPixelGameEngine PUBLIC OLC_IMAGE_STB)
    target_link_libraries(olcPixelGameEngine PRIVATE stb)
    target_link_libraries(olcPixelGameEngine PUBLIC X11::X11 OpenGL::GL)

    if(TARGET OpenGL::GLX)
        target_link_libraries(olcPixelGameEngine PUBLIC OpenGL::GLX)
    endif()
endif()
